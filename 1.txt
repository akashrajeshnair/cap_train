a small ecommerce company wants to analyze daily sales data.

each order contains orderid, customer_name and order_amount.

- has to store multiple orders
- has to calculate the total number of sales
- has to find the customer who has spent the most money
- has to display the orders with value more than 1000

data = [[1, "Akash", 5000],[2, "Raj", 400], [3, "Tom", 2400], [4, "John", 700]]

def calculate_total_sales(data):
    total = 0
    for d in data:
        total += d[2]
    return total
    
def find_max_spender(data):
    greatest, greatestpos = 0, 0
    for d in data:
        if d[2] > greatest:
            greatest = d[2]
            greatestpos = d[0]
    return data[greatestpos-1][1]

def find_orders_above_1000(data):
    orders = []
    for d in data:
        if d[2] > 1000:
            orders.append(d)
    return orders

print("total sales:")
print(calculate_total_sales(data))
print("max spender:")
print(find_max_spender(data))
print("orders above 1000")
print(find_orders_above_1000(data))
