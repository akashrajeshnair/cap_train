# password strength checker
print("Enter a password:")
password = input()
number = False

for p in password:
    if p.isnumeric():
        number = True

if len(password) >= 8:
    if number:
        print("Password is Strong.")
    else:
        print("Password is weak: No number.")
else:
    print("Password is weak: Too short.")

# palidrome checker

print("Enter a string:")
string = input()

string = string.lower()

if string == string[::-1]:
    print("Yes, it is a palindrome.")
else:
    print("No, it is not a palindrome.")
